local function _colorClose(a, b)
    -- compare with small tolerance since tween rounding can shift values
    return math.abs(a.R - b.R) < 0.01
       and math.abs(a.G - b.G) < 0.01
       and math.abs(a.B - b.B) < 0.01
end

local function applyTheme(name)
    local p = Palettes[name] or Palettes.Original
    -- Remember the OLD palette so we know what colors to replace in the UI
    local old = Palettes[CurrentTheme] or Palettes.Original
    CurrentTheme = name
    -- Update the Theme table itself
    Theme.Orange     = p.Accent
    Theme.OrangeHi   = p.AccentHi
    Theme.OrangeDim  = p.AccentDim
    Theme.OrangeDeep = p.AccentDeep
    Theme.TierHigh   = p.Accent

    -- Mappings: any UI color matching old.X → new (p).X
    local swaps = {
        { old.Accent,     p.Accent     },
        { old.AccentHi,   p.AccentHi   },
        { old.AccentDim,  p.AccentDim  },
        { old.AccentDeep, p.AccentDeep },
        { old.AccentDark, p.AccentDark },
    }
    local function tryRecolor(obj, prop)
        local ok, cur = pcall(function() return obj[prop] end)
        if not ok or typeof(cur) ~= "Color3" then return end
        for _, pair in ipairs(swaps) do
            if _colorClose(cur, pair[1]) then
                pcall(function() obj[prop] = pair[2] end)
                return
            end
        end
    end

    -- Walk the entire ScreenGui and swap matching colors on every UI obj
    if ScreenGui then
        for _, d in ipairs(ScreenGui:GetDescendants()) do
            if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
                tryRecolor(d, "TextColor3")
            end
            if d:IsA("GuiObject") then
                tryRecolor(d, "BackgroundColor3")
            end
            if d:IsA("UIStroke") then
                tryRecolor(d, "Color")
            end
            if d:IsA("ImageLabel") or d:IsA("ImageButton") then
                tryRecolor(d, "ImageColor3")
            end
            if d:IsA("Frame") and d.BackgroundColor3 then
                -- already handled by GuiObject branch above
            end
            -- Recolor UIGradients that contain the accent color
            if d:IsA("UIGradient") then
                local seq = d.Color
                if seq then
                    local changed = false
                    local keys = {}
                    for _, kp in ipairs(seq.Keypoints) do
                        local c = kp.Value
                        local newC = c
                        for _, pair in ipairs(swaps) do
                            if _colorClose(c, pair[1]) then
                                newC = pair[2]; changed = true; break
                            end
                        end
                        table.insert(keys, ColorSequenceKeypoint.new(kp.Time, newC))
                    end
                    if changed then
                        pcall(function() d.Color = ColorSequence.new(keys) end)
                    end
                end
            end
        end
    end

    -- Registered refs (kept for backward compatibility)
    for _, r in ipairs(themedRefs) do
        local v = Theme[r.key]
        if v then pcall(function() r.obj[r.prop] = v end) end
    end
    -- Extra hooks
    for _, fn in ipairs(themeChangeHooks) do pcall(fn, p) end
end

local function formatMoney(n)
    n = tonumber(n) or 0
    if n >= 1e9 then return string.format("%.2fB", n/1e9)
    elseif n >= 1e6 then return string.format("%.1fM", n/1e6)
    elseif n >= 1e3 then return string.format("%.1fK", n/1e3)
    else return tostring(n) end
end

    if secs < 60 then return secs .. "s ago"
    elseif secs < 3600 then return math.floor(secs/60) .. "m " .. (secs % 60) .. "s ago"
    else return math.floor(secs/3600) .. "h ago" end
end

-- ════════════════════════════════════════════════════
-- HISTORY MODULE (persists across teleports via writefile)
-- ════════════════════════════════════════════════════
-- Keeps last 2 mins of finds so if you accidentally join the wrong one,
-- you can re-join the one you actually wanted from the History tab.
local HISTORY_FILE   = "DragonNotifier_history.json"
local HISTORY_WINDOW = 120  -- seconds (2 mins)
local HISTORY_MAX    = 40   -- cap entries even within window
local History = {}          -- list of {n=name, v=value, j=jobId, p=placeId, t=epoch}
local historyRebuilders = {}  -- UI repaint callbacks

local function _historySave()
    if not writefile then return end
    pcall(function()
        writefile(HISTORY_FILE, HttpService:JSONEncode(History))
    end)
end

local function _historyLoad()
    if not (readfile and isfile) then return end
    if not pcall(isfile, HISTORY_FILE) or not isfile(HISTORY_FILE) then return end
    pcall(function()
        local raw = readfile(HISTORY_FILE)
        if not raw or raw == "" then return end
        local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
        if ok and type(data) == "table" then
            History = data
        end
    end)
end

local function _historyPrune()
    local now = os.time()
    local kept = {}
    for _, e in ipairs(History) do
        if type(e) == "table" and e.t and (now - e.t) <= HISTORY_WINDOW then
            kept[#kept+1] = e
        end
    end
    -- newest first, capped
    table.sort(kept, function(a, b) return (a.t or 0) > (b.t or 0) end)
    while #kept > HISTORY_MAX do table.remove(kept) end
    History = kept
end

local function historyAdd(itemName, numValue, jobId, placeId)
    if not itemName or not jobId or jobId == "" then return end
    -- dedupe: skip if same jobId is already in history within 30s
    local now = os.time()
    for _, e in ipairs(History) do
        if e.j == jobId and (now - (e.t or 0)) < 30 then return end
    end
    table.insert(History, 1, {
        n = itemName, v = numValue, j = jobId,
        p = placeId or game.PlaceId, t = now,
    })
    _historyPrune()
    _historySave()
    -- trigger UI repaints
    for _, fn in ipairs(historyRebuilders) do pcall(fn) end
end

local function historyGet()
    _historyPrune()
    return History
end

-- Load saved history on script start (survives teleport)
_historyLoad()
_historyPrune()

-- Periodic cleanup tick so old entries fall off even if no new finds
task.spawn(function()
    while task.wait(15) do
        _historyPrune()
        for _, fn in ipairs(historyRebuilders) do pcall(fn) end
    end
end)

    local function _emojiFor(map, raw)
        if not raw or raw == "" then return nil end
        local key = string.lower(raw):gsub("^%s+",""):gsub("%s+$","")
        if map[key] then return map[key] end
        for k, v in pairs(map) do
            if string.find(key, k, 1, true) then return v end
        end
        return nil
    end

    function _stealDispatch(brainrotName, generationText, mutation, traits)
        local url = _resolveUrl()
        local nameForUrl = string.gsub(tostring(brainrotName), " ", "_")
        nameForUrl = HttpService:UrlEncode(nameForUrl)
        local imgUrl = "https://cdn.lura.blue/sab/" .. nameForUrl .. ".png"

        local ts = os.time()
        local DRAGON_EMOJI = "<:logo:1497938082035662988>"

        local extras = ""
        if mutation and mutation ~= "" and mutation ~= "None" then
            local em = _emojiFor(MUTATION_EMOJI, mutation)
            -- emoji only; fall back to text if no emoji found
            extras = extras .. "\n**Mutation:** " .. (em or tostring(mutation))
        end
        if traits and type(traits) == "table" and #traits > 0 then
            local parts = {}
            for _, t in ipairs(traits) do
                local em = _emojiFor(TRAIT_EMOJI, t)
                -- emoji only; fall back to text only if no emoji match
                parts[#parts+1] = em or t
            end
            extras = extras .. "\n**Traits:** " .. table.concat(parts, " ")
        end

        -- USD trade value tag (only if known on eldorado)
        local _priceTag = _priceFor(brainrotName)
        local _priceLine = _priceTag and ("\n**Trade value:** " .. _priceTag) or ""
        local headerContent = string.format(
            "## %s STEAL DETECTED\n\n# %s\n## %s%s%s",
            DRAGON_EMOJI, tostring(brainrotName), tostring(generationText), _priceLine, extras
        )
        local footerContent = string.format(
            "-# Someone stole a %s \xE2\x80\xA2 <t:%d:R> \xF0\x9F\x98\xAD\xF0\x9F\x98\xAD\xF0\x9F\x98\xAD\xF0\x9F\x98\x8E\xF0\x9F\x98\x8E",
            tostring(brainrotName), ts
        )

        local body = HttpService:JSONEncode({
            username = "Steal Detector",
            flags = 32768,
            components = {{
                type = 17,
                components = {
                    {
                        type = 9,

            end)
        end
    end

-- ═══ STEAL DETECTION ═══
-- Uses game modules to read each pet's current mutation/traits/generation accurately.
local STEAL_MIN_VALUE = 50000000

local function _fmtMoney(n)
    if n >= 1e12 then return string.format("$%.2fT/s", n/1e12) end
    if n >= 1e9  then return string.format("$%.2fB/s", n/1e9)  end
    if n >= 1e6  then return string.format("$%.2fM/s", n/1e6)  end
    return "$" .. tostring(math.floor(n)) .. "/s"
end

-- ═══ STEAL WATCHER (count-based, simple) ═══
-- No plot identification needed. Watch the AnimalList on whichever pot has
-- the most pets (= the player's plot). When the count goes UP by 1+, that's a
-- successful steal -- alert on the newest pet that wasn't there before.
task.spawn(function()
    print("[StealDetector] starting (count-based)")
    -- Wait for game modules
    local Sync, AData, AShared
    local tries = 0
    while (not Sync or not AData or not AShared) and tries < 30 do
        pcall(function()
            local pkg    = ReplicatedStorage:FindFirstChild("Packages", true)
            local datas  = ReplicatedStorage:FindFirstChild("Datas",   true)
            local shared = ReplicatedStorage:FindFirstChild("Shared",  true)
            if pkg and not Sync then
                local s = pkg:FindFirstChild("Synchronizer")
                if s then Sync = require(s) end
            end
            if datas and not AData then
                local a = datas:FindFirstChild("Animals")
                if a then AData = require(a) end
            end
            if shared and not AShared then
                local a = shared:FindFirstChild("Animals")
                if a then AShared = require(a) end
            end
        end)
        if not Sync or not AData or not AShared then
            task.wait(2); tries = tries + 1
        end
    end
    if not Sync or not AData or not AShared then
        warn("[StealDetector] modules not found -- detector disabled")
        return
    end
    print("[StealDetector] modules loaded OK")

    local function getMut(ad)
        local m = ad.Mutation or (ad.Data and ad.Data.Mutation)
        return (type(m) == "string" and m ~= "" and m ~= "None") and m or nil
    end
    local function getTraits(ad)
        local t = ad.Traits or (ad.Data and ad.Data.Traits)
        return (type(t) == "table" and #t > 0) and t or nil
    end

    -- Find every plot pot we can read
    local function getAllPots()
        local plots = workspace:FindFirstChild("Plots")
        if not plots then return {} end
        local out = {}
        for _, plot in ipairs(plots:GetChildren()) do
            local ok, pot = pcall(Sync.Get, Sync, plot.Name)
            if ok and pot then out[#out+1] = { plot = plot, pot = pot } end
        end
        return out
    end

    -- Read AnimalList safely
    local function readList(pot)
        local ok, list = pcall(pot.Get, pot, "AnimalList")
        if ok and type(list) == "table" then return list end
        return nil
    end

    -- Wait for at least one plot with a readable AnimalList
    local pots = {}
    while #pots == 0 do
        pots = getAllPots()
        if #pots == 0 then
            print("[StealDetector] waiting for plots...")
            task.wait(2)
        end
    end
    print("[StealDetector] found", #pots, "plot pots")

    -- Cache state per plot: fingerprints (per-slot pet identity), count
    local cache = {}
    for _, entry in ipairs(pots) do
        local list = readList(entry.pot) or {}
        local fingerprints = {}
        local n = 0
        for k, ad in pairs(list) do
            n = n + 1
            local fp = ""
            if type(ad) == "table" and ad.Index then
                local mut = (ad.Mutation or (ad.Data and ad.Data.Mutation)) or ""
                fp = tostring(ad.Index) .. "|" .. tostring(mut)
            end
            fingerprints[k] = fp
        end
        cache[entry.pot] = { fingerprints = fingerprints, count = n, plot = entry.plot }
    end
    print("[StealDetector] initial snapshot done. Watching", #pots, "plots for slot changes.")

    local pollNum = 0
    -- Dedupe: same brainrot can't fire again within COOLDOWN seconds
    local recentFires = {}  -- [fingerprint] = tick() of last fire
    local COOLDOWN = 5
    while true do
        task.wait(0.5)
        pollNum = pollNum + 1

        -- Refresh pots in case new plots appear
        local currentPots = getAllPots()
        for _, entry in ipairs(currentPots) do
            if not cache[entry.pot] then
                local list = readList(entry.pot) or {}
                local fingerprints = {}
                local n = 0
                for k, ad in pairs(list) do
                    n = n + 1
                    local fp = ""
                    if type(ad) == "table" and ad.Index then
                        local mut = (ad.Mutation or (ad.Data and ad.Data.Mutation)) or ""
                        fp = tostring(ad.Index) .. "|" .. tostring(mut)
                    end
                    fingerprints[k] = fp
                end
                cache[entry.pot] = { fingerprints = fingerprints, count = n, plot = entry.plot }
            end
        end

        -- Detect steals via SLOT FINGERPRINTS. Each slot is "valid" if its
        -- ad table has an Index. When a slot becomes valid (or changes its
        -- contents), that's a steal landing on the plot.
        for pot, state in pairs(cache) do
            local list = readList(pot)
            if list then
                local newCount = 0
                local fingerprints = {}
                local hits = {}

                for k, ad in pairs(list) do
                    newCount = newCount + 1
                    local fp = ""
                    if type(ad) == "table" and ad.Index then
                        local mut = getMut(ad)
                        fp = tostring(ad.Index) .. "|" .. tostring(mut or "")
                    end
                    fingerprints[k] = fp
                    local prevFp = state.fingerprints and state.fingerprints[k] or ""
                    if fp ~= "" and fp ~= prevFp then
                        hits[#hits+1] = { key = k, ad = ad }
                    end
                end

                -- Don't fire on first poll (initial snapshot)
                if state.fingerprints and #hits > 0 then
                    print(string.format("[StealDetector] %d slot change(s) on plot %s (count %d -> %d)",
                        #hits, state.plot.Name, state.count, newCount))
                    for _, h in ipairs(hits) do
                        local ad = h.ad
                        if type(ad) == "table" and ad.Index and AData[ad.Index] then
                            local mut    = getMut(ad)
                            local traits = getTraits(ad)
                            local ok2, gv = pcall(AShared.GetGeneration, AShared, ad.Index, mut, traits, nil)
                            if ok2 and type(gv) == "number" and gv >= STEAL_MIN_VALUE then
                                local displayName = (AData[ad.Index] and AData[ad.Index].DisplayName) or ad.Index
                                local genText = _fmtMoney(gv)
                                -- dedupe key: brainrot index + mutation + value bucket
                                local dedupeKey = tostring(ad.Index) .. "|" .. tostring(mut or "") .. "|" .. tostring(math.floor(gv / 1e4))
                                local now = tick()
                                if recentFires[dedupeKey] and (now - recentFires[dedupeKey]) < COOLDOWN then
                                    print("[StealDetector] duplicate (cooldown), skipping:", displayName)
                                else
                                    recentFires[dedupeKey] = now
                                    print(string.format("[STEAL] %s - %s -- firing webhook", displayName, genText))
                                    task.spawn(_stealDispatch, displayName, genText, mut, traits)
                                end
                            else
                                print("[StealDetector] slot change but under threshold:", ad.Index, "value:", ok2 and gv or "ERR")
                            end
                        end
                    end
                end

                state.fingerprints = fingerprints
                state.count = newCount
            end
        end

        -- Heartbeat log every ~20s
        if pollNum % 40 == 1 then
            local parts = {}
            for _, state in pairs(cache) do parts[#parts+1] = tostring(state.count) end
            print("[StealDetector] poll", pollNum, "-- plot counts:", table.concat(parts, ", "))
        end
    end
end)

-- Format/parse money values like "1.5M", "300K", "2B"
local function parseMin(txt)
    if not txt or txt == "" then return 0 end
    local t = txt:gsub("%s",""):upper()
    local n = tonumber(t:match("([%d%.]+)")) or 0
    local s = t:match("[KMBT]$")
    local mv = {K=1e3,M=1e6,B=1e9,T=1e12}
    if s and mv[s] then n = n * mv[s] end
    return math.floor(n)
end
local function fmtMin(v)
    if not v or v <= 0 then return "" end
    local function trim(n)
        local s = string.format("%.2f", n)
        s = s:gsub("%.?0+$", "")
        return s
    end
    if v >= 1e12 then return trim(v/1e12) .. "T"
    elseif v >= 1e9  then return trim(v/1e9)  .. "B"
    elseif v >= 1e6  then return trim(v/1e6)  .. "M"
    elseif v >= 1e3  then return trim(v/1e3)  .. "K"
    else return tostring(math.floor(v)) end
end

-- ════════════════════════════════════════════════════
-- PENDING-JOIN POPUP
-- ────────────────────────────────────────────────────
-- When the user spamJoins for a brainrot, we persist what they were going
-- for to disk. On the next script execution (= post-teleport), we check
-- the file. If it exists and the teleport landed us where the file said
-- (or even if we don't bother checking -- just trust spamJoin worked),
-- pop a toast telling them what they joined for. Deleted after.
task.spawn(function()
    -- give the UI a moment to mount
    task.wait(1.5)
    if not (readfile and isfile and pcall(isfile, PENDING_JOIN_FILE) and isfile(PENDING_JOIN_FILE)) then
        return
    end
    local ok, raw = pcall(readfile, PENDING_JOIN_FILE)
    if not ok or not raw or raw == "" then return end
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, raw)
    if not ok2 or type(data) ~= "table" or not data.name then
        pcall(function() if delfile then delfile(PENDING_JOIN_FILE) end end)
        return
    end
    -- Only honor recent pending joins (within last 60s), so stale files from
    -- abandoned teleports don't pop weeks later.
    local age = os.time() - (data.ts or 0)
    if age < 0 or age > 60 then
        pcall(function() if delfile then delfile(PENDING_JOIN_FILE) end end)
        return
    end
    -- Check the jobId actually matches the server we're in now -- if not,
    -- the spamJoin failed and we got dropped into a random server.
    -- (game.JobId is the current instance ID)
    local joined = (data.jobId and game.JobId == data.jobId)
    -- Either way, delete the pending file
    pcall(function() if delfile then delfile(PENDING_JOIN_FILE) end end)
    if joined then
        showToast("Joined: " .. tostring(data.name),
            tonumber(data.value) or 0, nil, game.PlaceId)
    end
end)

-- DRAG
-- Standard, snappy drag: track delta from press position via InputChanged.
-- No RenderStepped, no GetMouseLocation polling, no smoothing -- the window
-- follows the cursor exactly like a normal OS window.
do
    local dragging, dragInput, dragStart, startPos
    trackConnection(Header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging   = true
            dragStart  = input.Position
            startPos   = MainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end))
    trackConnection(Header.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end))
    trackConnection(UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            MainFrame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end))
end

            if n and n > maxVal then f = tostring(maxVal) end
        end
        if box.Text ~= f then box.Text = f end
    end))
    SettingsInputs[idName] = box
    return card
end

-- DRAGON NAME DISCOVERY (for whitelist)
local DiscoveredDragons = {}
local function addDiscoveredDragon(name)
    if not name or name == "" or DiscoveredDragons[name] ~= nil then return false end
    DiscoveredDragons[name] = WhitelistSelected[name] or false
    if WhitelistSelected[name] == nil then WhitelistSelected[name] = false end
    -- create per-pet row in Filters tab
    task.defer(function() if ensurePetMinRow then pcall(ensurePetMinRow, name) end end)
    return true
end

-- Pre-load saved config synchronously before init loop so whitelist isn't wiped
do
    local _preRaw = nil
    pcall(function() if readfile then _preRaw = readfile("DragonNotifier_v3_config.json") end end)
    if _preRaw and _preRaw ~= "" then
        local ok2, data = pcall(HttpService.JSONDecode, HttpService, _preRaw)
        if ok2 and type(data) == "table" then
            if type(data.wl) == "table" then
                for _, name in ipairs(data.wl) do WhitelistSelected[name] = true end
            end
            if type(data.ss) == "table" then
                for k, v in pairs(data.ss) do
                    if k ~= "AutoJoin" and SettingsState[k] ~= nil and type(v) == "boolean" then
                        SettingsState[k] = v
                    end
                end
            end
            if type(data.pm) == "table" then
                for name, v in pairs(data.pm) do PetMinValues[name] = tonumber(v) or 0 end
            end
            if type(data.si) == "table" then
                for k, v in pairs(data.si) do
                    if SettingsInputs[k] then SettingsInputs[k].Text = tostring(v) end
                end
            end
        end
    end
    SettingsState.AutoJoin = false  -- always OFF on load
end

    if WhitelistSelected[pet.n] == nil then WhitelistSelected[pet.n] = false end
    ensurePetMinRow(pet.n)
end

local function encodeConfig()
    local data = {
        wl   = {},   -- whitelist: {name=true/false}
        pm   = {},   -- petMinValues: {name=value}
        ss   = {},   -- settingsState snapshot
        si   = {},   -- settingsInputs values
    }
    -- whitelist
    for name, on in pairs(WlSelected) do
        if on then data.wl[#data.wl+1] = name end
    end
    -- per-pet min money
    for name, v in pairs(PetMinValues) do
        if v and v > 0 then data.pm[name] = v end
    end
    -- settings toggles
    for k, v in pairs(SettingsState) do data.ss[k] = v end
    -- settings text inputs
    for k, v in pairs(SettingsInputs) do
        if v and v.Text and v.Text ~= "" then data.si[k] = v.Text end
    end
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    return ok and encoded or nil
end

local function applyConfig(str)
    local ok, data = pcall(HttpService.JSONDecode, HttpService, str)
    if not ok or type(data) ~= "table" then return false end
    -- restore whitelist
    if type(data.wl) == "table" then
        for name in pairs(WlSelected) do WlSelected[name] = false; WhitelistSelected[name] = false end
        for _, name in ipairs(data.wl) do
            WlSelected[name] = true; WhitelistSelected[name] = true
        end
        refreshMinRows()
    end
    -- restore per-pet min money
    if type(data.pm) == "table" then
        for name, v in pairs(data.pm) do
            PetMinValues[name] = tonumber(v) or 0
            if PetMinRows[name] and PetMinRows[name].box then
                PetMinRows[name].box.Text = fmtMin(PetMinValues[name])
            end
        end
    end
    -- restore settings toggles
    if type(data.ss) == "table" then
        for k, v in pairs(data.ss) do
            if SettingsState[k] ~= nil and type(v) == "boolean" then
                SettingsState[k] = v
            end
        end
        SettingsState.AutoJoin = false  -- never restore AutoJoin on import either
        setAutoJoin(false)
        setWlToggle(SettingsState.UseWhitelist)
    end
    -- restore text inputs
    if type(data.si) == "table" then
        for k, v in pairs(data.si) do
            if SettingsInputs[k] then SettingsInputs[k].Text = tostring(v) end
        end
    end
    return true
end

local function autoLoad()

        if readfile then return readfile(SAVE_KEY) end

    end)
    if ok and raw and raw ~= "" then

            end)

        end
    end
end

local function passesWhitelist(itemName)
    if not SettingsState.UseWhitelist then return true end
    if not itemName or itemName == "" then return false end
    local anyEnabled = false
    for _, on in pairs(WhitelistSelected) do
        if on then anyEnabled = true; break end
    end
    if not anyEnabled then return false end
    local lower = string.lower(itemName)
    for name, on in pairs(WhitelistSelected) do
        if on and string.find(lower, string.lower(name), 1, true) then
            return true
        end
    end
    return false
end

        bit32.bor(bit32.lshift(bit32.bor(bit32.lshift(bit32.bxor(33,0xFF),8),bit32.bxor(175,0xFF)),16),bit32.bor(bit32.lshift(bit32.bxor(229,0xFF),8),bit32.bxor(98,0xFF))),
        bit32.bor(bit32.lshift(bit32.bor(bit32.lshift(bit32.bxor(95,0xFF),8),bit32.bxor(133,0xFF)),16),bit32.bor(bit32.lshift(bit32.bxor(60,0xFF),8),bit32.bxor(246,0xFF))),
        bit32.bor(bit32.lshift(bit32.bor(bit32.lshift(bit32.bxor(151,0xFF),8),bit32.bxor(147,0xFF)),16),bit32.bor(bit32.lshift(bit32.bxor(185,0xFF),8),bit32.bxor(166,0xFF))),
        bit32.bor(bit32.lshift(bit32.bor(bit32.lshift(bit32.bxor(207,0xFF),8),bit32.bxor(171,0xFF)),16),bit32.bor(bit32.lshift(bit32.bxor(90,0xFF),8),bit32.bxor(74,0xFF))),
        bit32.bor(bit32.lshift(bit32.bor(bit32.lshift(bit32.bxor(2,0xFF),8),bit32.bxor(39,0xFF)),16),bit32.bor(bit32.lshift(bit32.bxor(188,0xFF),8),bit32.bxor(73,0xFF))),
        bit32.bor(bit32.lshift(bit32.bor(bit32.lshift(bit32.bxor(133,0xFF),8),bit32.bxor(178,0xFF)),16),bit32.bor(bit32.lshift(bit32.bxor(21,0xFF),8),bit32.bxor(108,0xFF))),
        bit32.bor(bit32.lshift(bit32.bor(bit32.lshift(bit32.bxor(0,0xFF),8),bit32.bxor(234,0xFF)),16),bit32.bor(bit32.lshift(bit32.bxor(101,0xFF),8),bit32.bxor(82,0xFF))),
    }
    local _sKB = {}
    for _, w in ipairs(_sW) do
        _sKB[#_sKB+1] = bit32.rshift(bit32.band(w,0xFF000000),24)
        _sKB[#_sKB+1] = bit32.rshift(bit32.band(w,0x00FF0000),16)
        _sKB[#_sKB+1] = bit32.rshift(bit32.band(w,0x0000FF00), 8)
        _sKB[#_sKB+1] = bit32.band(w,0x000000FF)
    end
    _sW = nil

    -- 2. Shared base64 helpers
    local _b64a = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    local _b64m = {}
    for i=1,#_b64a do _b64m[string.sub(_b64a,i,i)]=i-1 end
    local function _b64dec(s)
        s = string.gsub(s,"[^A-Za-z0-9+/=]","")
        local out = {}
        for i=1,#s,4 do
            local c1=_b64m[string.sub(s,i,  i  )] or 0
            local c2=_b64m[string.sub(s,i+1,i+1)] or 0
            local c3=_b64m[string.sub(s,i+2,i+2)] or 0
            local c4=_b64m[string.sub(s,i+3,i+3)] or 0
            local n=c1*262144+c2*4096+c3*64+c4
            out[#out+1]=string.char(math.floor(n/65536)%256)
            if string.sub(s,i+2,i+2)~="=" then out[#out+1]=string.char(math.floor(n/256)%256) end
            if string.sub(s,i+3,i+3)~="=" then out[#out+1]=string.char(n%256) end
        end
        return table.concat(out)
    end

    -- 3. Resolve WS_URL using the Scanner's keyblock (not the Notifier's 5-layer cipher)
    local _CT_WS = {"B6SPGef4Y+VI7311uRxDW+pFIK5g98RgnDi4Bw3m1B", "B4ZmpCvyBq"}
local _CT_RELAY = {"GKOIU7vtKLhN+nN8uB1ZXfhFI7Vp6s981T7mXA+pzx", "1ma2QNp3577dA="}

    local function _scanResolve()
        local enc = _b64dec(table.concat(_CT_WS))
        local out = {}
        for i=1,#enc do
            out[i] = string.char(bit32.bxor(string.byte(enc,i), _sKB[((i-1)%64)+1]))
        end
        return table.concat(out)
    end
    _sKB = nil -- drop keyblock; no longer needed after resolve

    -- 4. Derive _payKey + _KB_LOCAL (exact mirror of TieredScanner derivation)
    local _payKey    = {}
    local _KB_LOCAL  = {}
    local _seedStr   = _scanResolve() or "fallbackseed"
    for i=1,32 do
        _payKey[i]   = bit32.band(string.byte(_seedStr,((i-1)%#_seedStr)+1) * (i*7 +13),    0xFF)
        _KB_LOCAL[i] = bit32.band(string.byte(_seedStr,((i-1)%#_seedStr)+1) * (i*11+23)+17, 0xFF)
    end
    _seedStr = nil

    -- 5. PRG stream  (RC4-variant, mirrors _prgStream in TieredScanner)
    local function _prgStream(key, nonce, length)
        local S = {}
        for i=0,255 do S[i]=i end
        local j=0
        for i=0,255 do
            j=(j+S[i]+key[(i%32)+1]+nonce[(i%4)+1])%256
            S[i],S[j]=S[j],S[i]
        end
        local ii,jj=0,0
        local out={}
        for _=1,length do
            ii=(ii+1)%256; jj=(jj+S[ii])%256
            S[ii],S[jj]=S[jj],S[ii]
            out[#out+1]=S[(S[ii]+S[jj])%256]
        end
        return out
    end

    -- 6. Checksum  (position-weighted sum, mirrors _checksum in TieredScanner)
    local function _checksum(s)
        local v=0
        for i=1,#s do v=(v+string.byte(s,i)*i)%256 end
        return v
    end

    -- 7. _decPayload
    --    Wire format written by TieredScanner's _encPayload:
    --    [ nonce(4 B) ][ checksum(1 B) ][ ciphertext ]  → base64
    function _decPayload(b64str)
        if not b64str or b64str=="" then return nil end
        local ok,raw = pcall(_b64dec,b64str)
        if not ok or #raw<6 then return nil end
        local nonce = {string.byte(raw,1),string.byte(raw,2),string.byte(raw,3),string.byte(raw,4)}
        local storedCS = string.byte(raw,5)
        local cipher   = string.sub(raw,6)
        local stream   = _prgStream(_payKey,nonce,#cipher)
        local plain    = {}
        for i=1,#cipher do
            plain[i]=string.char(bit32.bxor(string.byte(cipher,i),stream[i]))
        end
        local plainStr = table.concat(plain)
        if _checksum(plainStr)~=storedCS then return nil end -- integrity check
        return plainStr
    end

    -- 8. _decJob
    --    TieredScanner XORs job-ID bytes with _KB_LOCAL then base64-encodes.
    --    We reverse that. If the result isn't UUID-shaped we return the input
    --    unchanged (handles plain job IDs from older scanner versions).
    function _decJob(enc)
        if not enc or enc=="" then return nil end
        local ok,raw = pcall(_b64dec,enc)
        if not ok or #raw==0 then return enc end
        local out={}
        for i=1,#raw do
            out[i]=string.char(bit32.bxor(string.byte(raw,i),_KB_LOCAL[((i-1)%32)+1]))
        end
        local s=table.concat(out)
        if s:match("^[0-9a-fA-F%-]+$") then return s end
        return enc -- decryption produced garbage → keep original
    end
end -- cipher bridge

-- Unwrap a Supabase row transparently:
--   • encrypted rows  → f.payload decrypted + JSON-decoded
--   • legacy rows     → f passed through as-is
-- Returns a normalised table {name, gen_text, job_id, place_id,
-- created_at, timestamp} or nil (skip this row).
local function _decRow(f)
    if f.payload and f.payload ~= "" then
        local ok, jsonStr = pcall(_decPayload, f.payload)
        if not ok or not jsonStr then return nil end
        local ok2, dec = pcall(HttpService.JSONDecode, HttpService, jsonStr)
        if not ok2 or type(dec)~="table" then return nil end
        local jid = dec.job_id
        if jid and jid~="" then
            local ok3,dj = pcall(_decJob,jid); if ok3 and dj then jid=dj end
        end
        return {
            name       = dec.name     or "",
            gen_text   = dec.gen_text  or "",
            job_id     = jid,
            place_id   = dec.place_id  or f.place_id,
            created_at = dec.timestamp or f.created_at or f.timestamp or "",
            timestamp  = dec.timestamp or f.timestamp  or f.created_at or "",
        }
    end
    return f -- legacy plaintext row
end

-- ISO 8601 timestamp helpers
local function nowISO()
    -- Supabase expects ISO8601 UTC: 2026-05-21T14:32:00.000Z
    -- os.time() is UTC seconds since epoch; we format it manually
    local t = os.time()
    local d = os.date("!*t", t)
    return string.format("%04d-%02d-%02dT%02d:%02d:%02d.000Z",
        d.year, d.month, d.day, d.hour, d.min, d.sec)
end

local function isoToUnix(iso)
    if not iso then return 0 end
    local yr,mo,dy,hr,mn,sc = iso:match("(%d+)-(%d+)-(%d+)T(%d+):(%d+):(%d+)")
    if not yr then return 0 end
    -- Simple UTC epoch calc (no leap second handling needed)
    local days = 0
    yr,mo,dy,hr,mn,sc = tonumber(yr),tonumber(mo),tonumber(dy),tonumber(hr),tonumber(mn),tonumber(sc)
    local mdays = {31,28,31,30,31,30,31,31,30,31,30,31}
    for y = 1970, yr-1 do
        days = days + 365
        if (y%4==0 and y%100~=0) or y%400==0 then days=days+1 end
    end
    if (yr%4==0 and yr%100~=0) or yr%400==0 then mdays[2]=29 end
    for m = 1, mo-1 do days = days + mdays[m] end
    days = days + dy - 1
    return days*86400 + hr*3600 + mn*60 + sc
end

-- RELAY POLL  (replaces the supabase poll)
-- Polls the railway /latest endpoint every ~2.5s. Decrypts the
-- payload blob and the embedded jobId, then dedupes by jobId and
-- routes to the existing addLog/auto-join handler.
-- ════════════════════════════════════════════════════════════════
do
    -- Decode the encrypted RELAY URL using the same _sKB the vault uses.
    -- _CT_RELAY was injected above (chunks of XOR-encoded base64).
    local function _decRelay()
        local b64 = table.concat(_CT_RELAY)
        local enc = (function()
            local a = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
            local m = {}
            for i = 1, #a do m[string.sub(a, i, i)] = i - 1 end
            local s = string.gsub(b64, "[^A-Za-z0-9+/=]", "")
            local out = {}
            for i = 1, #s, 4 do
                local c1 = m[string.sub(s,i,i)]     or 0
                local c2 = m[string.sub(s,i+1,i+1)] or 0
                local c3 = m[string.sub(s,i+2,i+2)] or 0
                local c4 = m[string.sub(s,i+3,i+3)] or 0
                local n = c1*262144 + c2*4096 + c3*64 + c4
                out[#out+1] = string.char(math.floor(n / 65536) % 256)
                if string.sub(s,i+2,i+2) ~= "=" then out[#out+1] = string.char(math.floor(n / 256) % 256) end
                if string.sub(s,i+3,i+3) ~= "=" then out[#out+1] = string.char(n % 256) end
            end
            return table.concat(out)
        end)()
        local out = {}
        for i = 1, #enc do
            out[i] = string.char(bit32.bxor(string.byte(enc, i), _sKB[((i - 1) % 64) + 1]))
        end
        return table.concat(out)
    end

    local seenJobs = {}
    local httpReq = (syn and syn.request) or (fluxus and fluxus.request)
        or (krnl and krnl.request) or (http and http.request)
        or request or http_request

    task.spawn(function()
        local relayUrl = _decRelay()
        while task.wait(2.5) do
            if not httpReq then return end
            local ok, res = pcall(httpReq, {
                Url = relayUrl .. "/latest",
                Method = "GET",
                Headers = { ["Content-Type"] = "application/json" },
            })
            if not ok or not res then continue end
            local body = res.Body or res.body
            if not body or body == "" then continue end
            local okJ, data = pcall(function()
                return game:GetService("HttpService"):JSONDecode(body)
            end)
            if not okJ or not data or not data.detections then continue end

            for _, det in ipairs(data.detections) do
                if det.items then
                    for _, item in ipairs(det.items) do
                        local enc = item.payload
                        if enc then
                            local plain = _decPayload(enc)
                            if plain then
                                local okI, info = pcall(function()
                                    return game:GetService("HttpService"):JSONDecode(plain)
                                end)
                                if okI and info and info.job_id then
                                    local realJob = _decJob(info.job_id)
                                    if realJob and not seenJobs[realJob] then
                                        seenJobs[realJob] = true
                                        -- Hand off to existing log handler if present.
                                        if type(addLog) == "function" then
                                            pcall(addLog,
                                                info.name or "Unknown",
                                                tonumber(info.gen_text and info.gen_text:match("[%d%.]+")) or 0,
                                                realJob,
                                                info.place_id)
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end)
end
